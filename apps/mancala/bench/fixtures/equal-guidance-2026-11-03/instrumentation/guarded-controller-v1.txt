const selectMatched = async function(gameNumber, ordinal, pit, seeds, fullLabel, validationOnly = false) {
  const root = '/private/tmp/sol-lfm3-matched-12';
  const manifestLocal = JSON.parse(await fs.readFile(root + '/manifest.json', 'utf8'));
  if (!Number.isInteger(gameNumber) || gameNumber < 1 || gameNumber > manifestLocal.plannedGames) throw Error('Invalid explicit game number');
  if (!Number.isInteger(ordinal) || ordinal < 1) throw Error('Invalid explicit Human ordinal');
  if (!Number.isInteger(pit) || pit < 1 || pit > 6 || !Number.isInteger(seeds) || seeds < 1) throw Error('Invalid explicit pit/seeds');
  const plan = manifestLocal.schedule[gameNumber - 1];
  if (plan.game !== gameNumber) throw Error('Manifest game mismatch');
  const dir = root + '/game-' + String(gameNumber).padStart(2, '0');
  const inputFile = dir + '/guidance/turn-' + String(ordinal).padStart(2, '0') + '.json';
  const decisionFile = dir + '/decisions.json';
  const guidance = JSON.parse(await fs.readFile(inputFile, 'utf8'));
  if (guidance.sessionId !== plan.humanSessionId) throw Error('Guidance session differs from manifest');
  const expectedLabel = 'Sow ' + seeds + (seeds === 1 ? ' seed' : ' seeds') + ' from AI pit ' + pit;
  if (fullLabel !== expectedLabel) throw Error('Explicit label differs from pit/seeds');
  const choice = guidance.input.choices.find(c => c.label === fullLabel);
  if (!choice || choice.id !== pit + 6) throw Error('Label absent from legal choices or wrong mirrored pit ID');
  const fresh = await tab.playwright.evaluate(() => {
    const board = Array(14).fill(null);
    const buttons = Array.from(document.querySelectorAll('.pit-grid button[aria-label]'));
    if (buttons.length !== 12) throw Error('Expected twelve visible pit controls');
    for (const b of buttons) {
      const m = b.getAttribute('aria-label').match(/^(Your|Opponent’s) pit (\d+), (\d+) seeds$/);
      if (!m) throw Error('Invalid visible pit label');
      board[m[1] === 'Your' ? +m[2] - 1 : +m[2] + 6] = +m[3];
    }
    board[6] = +document.querySelector('.store-human').getAttribute('aria-label').match(/(\d+) seeds/)[1];
    board[13] = +document.querySelector('.store-ai').getAttribute('aria-label').match(/(\d+) seeds/)[1];
    return {board, moveCount: document.querySelectorAll('.move-list > li').length, interactive: buttons.some(b => b.getAttribute('aria-label').startsWith('Your') && !b.disabled), heading: document.querySelector('main h1')?.textContent};
  });
  if (!fresh.interactive || !/^(Your move\. Make it count\.|Another handful\. Your turn\.)$/.test(fresh.heading)) throw Error('Human turn is not settled and interactive');
  if (fresh.board.some(n => !Number.isInteger(n) || n < 0) || fresh.board.reduce((a,b) => a+b,0) !== 48) throw Error('Visible board is incomplete');
  if (JSON.stringify(fresh.board) !== JSON.stringify(guidance.actualBoard)) throw Error('Visible board differs from exact guidance input');
  if (fresh.board[pit - 1] !== seeds) throw Error('Selected seed count differs from visible board');
  let rows;
  try { rows = JSON.parse(await fs.readFile(decisionFile, 'utf8')); } catch (e) { if (e.code === 'ENOENT') rows = []; else throw e; }
  if (!Array.isArray(rows)) throw Error('Decisions file is not an array');
  if (ordinal !== rows.length + 1 || rows.some(r => (r.humanTurnOrdinal ?? r.humanTurn) === ordinal || r.inputFile === inputFile)) throw Error('Duplicate or out-of-sequence Human ordinal/input path');
  if (rows.some(r => r.executed === false || r.status === 'pending')) throw Error('Unresolved pending decision');
  if (rows.length && rows[rows.length-1].moveNumber >= fresh.moveCount + 1) throw Error('Move number is not increasing');
  const row = {gameNumber, humanTurnOrdinal: ordinal, moveNumber: fresh.moveCount + 1, inputFile, sessionId: plan.humanSessionId, actualBoard: guidance.actualBoard, answer: {pit: fullLabel}, uiPit: pit, seeds, status: 'pending', executed: false, recordedAt: new Date().toISOString(), controllerVersion: 'explicit-game-v1'};
  if (validationOnly) return {validated: true, mutationPerformed: false, row};
  rows.push(row);
  await fs.writeFile(decisionFile, JSON.stringify(rows, null, 2));
  await tab.playwright.getByRole('button', {name: 'Your pit ' + pit + ', ' + seeds + ' seeds', exact: true}).click();
  const actual = await tab.playwright.evaluate(number => Array.from(document.querySelectorAll('.move-list > li')).map(e => e.innerText).find(t => Number(t.split('\n')[0]) === number), row.moveNumber);
  const parsed = actual?.match(/^(\d+)\n(You|Opponent)\nPit (\d+)\n(\d+) seeds/);
  if (!parsed || parsed[2] !== 'You' || +parsed[3] !== pit || +parsed[4] !== seeds) throw Error('Actual newly rendered Human move did not match pending decision; stop without retry');
  row.status = 'executed'; row.executed = true; row.actualMoveText = actual; row.executedAt = new Date().toISOString();
  await fs.writeFile(decisionFile, JSON.stringify(rows, null, 2));
  try { await tab.playwright.getByRole('heading', {name: /^(Your move\. Make it count\.|Another handful\. Your turn\.|A beautiful harvest\.|Perfectly balanced\.|A good game, well sown\.)$/, exact:true}).waitFor({state:'visible', timeoutMs:60000}); } catch (e) { nodeRepl.write('Move executed and logged; next turn still settling.'); }
  nodeRepl.write(await tab.playwright.evaluate(() => ({heading:document.querySelector('main h1')?.textContent,pits:Array.from(document.querySelectorAll('.pit-grid button[aria-label]')).map(b=>({label:b.getAttribute('aria-label'),enabled:!b.disabled})),stores:[document.querySelector('.store-human').getAttribute('aria-label'),document.querySelector('.store-ai').getAttribute('aria-label')],moveCount:document.querySelectorAll('.move-list > li').length})));
};

const finishMatched = async function(gameNumber, validationOnly = false) {
  const root = '/private/tmp/sol-lfm3-matched-12';
  const manifestLocal = JSON.parse(await fs.readFile(root + '/manifest.json','utf8'));
  if (!Number.isInteger(gameNumber) || gameNumber < 1 || gameNumber > manifestLocal.plannedGames) throw Error('Invalid explicit game number');
  const plan = manifestLocal.schedule[gameNumber-1];
  const dir = root + '/game-' + String(gameNumber).padStart(2,'0');
  const rows = JSON.parse(await fs.readFile(dir + '/decisions.json','utf8'));
  const ui = await tab.playwright.evaluate(() => ({origin:location.origin,heading:document.querySelector('main h1')?.textContent,history:Array.from(document.querySelectorAll('.move-list > li')).map(e=>e.innerText),renderedText:document.body.innerText,pitCounts:Array.from(document.querySelectorAll('.pit-grid button[aria-label]')).map(b=>+b.getAttribute('aria-label').match(/, (\d+) seeds$/)[1]),humanStore:+document.querySelector('.store-human').getAttribute('aria-label').match(/(\d+) seeds/)[1],lfmStore:+document.querySelector('.store-ai').getAttribute('aria-label').match(/(\d+) seeds/)[1]}));
  if (!/^(A beautiful harvest\.|Perfectly balanced\.|A good game, well sown\.)$/.test(ui.heading) || ui.pitCounts.length !== 12 || ui.pitCounts.some(n=>n!==0) || ui.humanStore+ui.lfmStore!==48) throw Error('UI does not show a completed 48-seed result');
  const moves=ui.history.map(t=>{const m=t.match(/^(\d+)\n(You|Opponent)\nPit (\d+)\n(\d+) seeds(?: · captured (\d+))?/);if(!m)throw Error('Unparseable rendered move');return{number:+m[1],player:m[2]==='You'?'human':'lfm',pit:+m[3],seeds:+m[4],captured:+(m[5]||0),extraTurn:t.includes('Go again'),handoff:!t.includes('Go again'),raw:t};}).sort((a,b)=>a.number-b.number);
  if (!moves.length || moves.some((m,i)=>m.number!==i+1) || moves[0].player!==(plan.startingSide==='human'?'human':'lfm')) throw Error('Move sequence or starting player mismatch');
  const humanMoves=moves.filter(m=>m.player==='human');
  if (!Array.isArray(rows) || rows.length!==humanMoves.length) throw Error('Decision count differs from actual Human moves');
  for (let i=0;i<rows.length;i++) {const r=rows[i],m=humanMoves[i];const expected=dir+'/guidance/turn-'+String(i+1).padStart(2,'0')+'.json';if((r.humanTurnOrdinal??r.humanTurn)!==i+1||r.inputFile!==expected||r.moveNumber!==m.number||r.uiPit!==m.pit||r.answer.pit!=='Sow '+m.seeds+(m.seeds===1?' seed':' seeds')+' from AI pit '+m.pit||r.executed===false||r.status==='pending')throw Error('Decision does not match exact Human move/input');const input=JSON.parse(await fs.readFile(expected,'utf8'));if(input.sessionId!==plan.humanSessionId)throw Error('Guidance session mismatch');}
  const result={...ui,gameNumber,startingSide:plan.startingSide,firstPlayer:plan.firstPlayer,humanSessionId:plan.humanSessionId,sourceCommit:manifestLocal.appSourceCommit,analysisSourceCommit:manifestLocal.analysisSourceCommit,promptVersion:14,opponent:'LFM3-1.7B-GGUF',opponentThinking:false,human:'GPT-6.1 Sol high',humanAssistance:'production-v14-mirrored',score:{human:ui.humanStore,lfm:ui.lfmStore},winner:ui.humanStore>ui.lfmStore?'human':ui.humanStore<ui.lfmStore?'lfm':'draw',moves,decisions:rows,decisionsFile:dir+'/decisions.json',completedAt:new Date().toISOString(),controllerVersion:'explicit-game-v1',instrumentationRecovery:gameNumber===2?{afterHumanOrdinal:9,description:'Original spillover preserved; first nine decisions transparently recovered before guarded explicit-game logging.'}:null};
  const progress=JSON.parse(await fs.readFile(root+'/progress.json','utf8'));
  if (progress.results.some(r=>r.gameNumber===gameNumber)) throw Error('Game already present in progress; preserve original instead of overwriting');
  if (validationOnly) return {validated:true,mutationPerformed:false,result};
  await fs.writeFile(dir+'/game.json',JSON.stringify(result,null,2));
  await fs.writeFile(dir+'/result.jpg',await tab.screenshot({fullPage:true}));
  progress.results.push({gameNumber,startingSide:result.startingSide,firstPlayer:result.firstPlayer,score:result.score,winner:result.winner,moves:moves.length,humanDecisions:rows.length,completedAt:result.completedAt,artifact:dir+'/game.json'});
  progress.completed=progress.results.length;progress.updatedAt=new Date().toISOString();
  await fs.writeFile(root+'/progress.json',JSON.stringify(progress,null,2));
  nodeRepl.write(progress.results[progress.results.length-1]);
};
